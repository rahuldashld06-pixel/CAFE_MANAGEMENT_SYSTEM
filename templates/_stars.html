{# Stars out of five for a rating, halves included. Read out as the number. #} {% macro stars(value, size="") -%} {%- if value is none -%} No stars {%- else -%} {%- for n in range(1, 6) -%} {%- if value >= n -%} {%- elif value >= n - 0.5 -%} {%- else -%}{%- endif -%} {%- endfor -%} {%- endif -%} {%- endmacro %}
